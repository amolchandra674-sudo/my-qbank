            const body = document.body;

                localStorage.setItem(PREFIX + 'theme', 'light');

                    savedSel = progress[q.id].sel;
                    showExpl = 'block';

                    <div class="opts-container ${lockClass}" id="opts-${q.id}">`;
                
                for(let optKey in q.options) {
                    let cClass = '';
                    if (isAnswered) {
                        if (optKey === q.ans) cClass = 'correct';
                        else if (optKey === savedSel && savedSel !== q.ans) cClass = 'wrong';
                    }
                    html += `

                    </div>
                </div>`;
            });
            document.getElementById('quiz-container').innerHTML = html;
        }

        window.handleSimpleAnswer = function(el, qId, correctAns, selectedOpt) {
            let container = document.getElementById('opts-' + qId);
            if (container.classList.contains('locked')) return;
            container.classList.add('locked');
            
            let isCorrect = (selectedOpt === correctAns);
            if (isCorrect) el.classList.add('correct');
            else {
                el.classList.add('wrong');
                Array.from(container.children).forEach(child => {
                    if (child.querySelector('.opt-lbl').innerText === correctAns) child.classList.add('correct');
                });
            }
            document.getElementById('expl-' + qId).style.display = 'block';

            if (currentContext === 'lesson') {
                progress[qId] = { status: isCorrect ? 'correct' : 'wrong', sel: selectedOpt };
                if (!isCorrect) mistakes[qId] = true; 
                else delete mistakes[qId]; 
                saveState();
            } else if (currentContext === 'mistake') {
                if (isCorrect) { delete mistakes[qId]; saveState(); }
            }
        }

            document.getElementById('main-bottom-nav').style.display = 'none';
            document.getElementById('lesson-reset-btn').style.display = 'none';
            document.getElementById('header-quiz-name').innerText = "Saved Bookmarks";
            let qList = Object.keys(bookmarks).map(id => Q_DICT[id]).filter(q => q);
            if (qList.length === 0) {
                document.getElementById('quiz-container').innerHTML = `<div class="empty-state"><i class="fas fa-bookmark"></i><h2>No Bookmarks</h2><p>Tap the bookmark icon on any question to save it here for review.</p></div>`;
                showScreen('screen-quiz');
                return;
            }
            renderSimpleView(qList);
            showScreen('screen-quiz');
        }

        window.openMistakes = function() {
            currentContext = 'mistake';
            document.getElementById('main-bottom-nav').style.display = 'none';
            document.getElementById('lesson-reset-btn').style.display = 'none';
            document.getElementById('header-quiz-name').innerText = "Review Mistakes";
            let qList = Object.keys(mistakes).map(id => Q_DICT[id]).filter(q => q);
            if (qList.length === 0) {
                document.getElementById('quiz-container').innerHTML = `<div class="empty-state"><i class="fas fa-check-circle" style="color:var(--green);"></i><h2>All Clear!</h2><p>You have no incorrect questions to review.</p></div>`;
                showScreen('screen-quiz');
                return;
            }
            renderSimpleView(qList);
            showScreen('screen-quiz');
        }

        window.toggleAllSubjects = function(cb) { document.querySelectorAll('.subj-chk').forEach(el => el.checked = cb.checked); }
        window.setMarking = function(style) {
            examStyle = style;
            document.getElementById('mark-neet').classList.remove('active');
            document.getElementById('mark-fmge').classList.remove('active');
            document.getElementById('mark-' + style.toLowerCase()).classList.add('active');
        }

        window.startCustomExam = function() {
            let selectedIdxs = Array.from(document.querySelectorAll('.subj-chk:checked')).map(cb => parseInt(cb.value));
            if(selectedIdxs.length === 0) return alert("Select at least one folder!");
            let numQ = parseInt(document.getElementById('ce-numq').value) || 50;
            let timeM = parseInt(document.getElementById('ce-time').value) || 50;
            
            let pool = [];
            selectedIdxs.forEach(idx => { APP_DATA[idx].lessons.forEach(l => pool.push(...l.questions)); });
            if(pool.length === 0) return alert("No questions in selected folders!");
            
            pool.sort(() => 0.5 - Math.random());
            let qList = pool.slice(0, numQ);
            startExamEngine(qList, timeM);
        }

        function startExamEngine(qList, timeMins) {
            activeExamQs = qList;
            activeExamAns = {};
            currentExamIdx = 0;
            timeRemainingSecs = timeMins * 60;
            
            document.getElementById('main-bottom-nav').style.display = 'none';
            renderExamQuestion();
            showScreen('screen-active-exam');
            
            clearInterval(timerInterval);
            updateTimerDisplay();
            timerInterval = setInterval(() => {
                timeRemainingSecs--;
                updateTimerDisplay();
                if(timeRemainingSecs <= 0) { clearInterval(timerInterval); submitExam(); }
            }, 1000);
        }

        function updateTimerDisplay() {
            let m = Math.floor(timeRemainingSecs / 60);
            let s = timeRemainingSecs % 60;
            document.getElementById('exam-timer').innerText = `${m.toString().padStart(2,'0')}:${s.toString().padStart(2,'0')}`;
        }

        function renderExamQuestion() {
            let q = activeExamQs[currentExamIdx];
            document.getElementById('exam-q-tracker').innerText = `Q ${currentExamIdx + 1} / ${activeExamQs.length}`;
            let imgHtml = q.img ? `<img src="${q.img}" class="q-img">` : '';
            let html = `
                <div class="q-card" style="box-shadow:none; border:none; padding:10px;">
                    <div class="q-text">${q.text} ${imgHtml}</div>
                    <div class="opts-container">`;
            for(let optKey in q.options) {
                let isSel = (activeExamAns[currentExamIdx] === optKey) ? 'selected' : '';
                html += `
                    <div class="option ${isSel}" onclick="selectExamOption('${optKey}')">
                        <div class="opt-lbl">${optKey}</div>
                        <div style="flex:1;">${q.options[optKey]}</div>
                    </div>`;
            }
            html += `</div></div>`;
            document.getElementById('exam-q-container').innerHTML = html;
        }

        window.selectExamOption = function(optKey) { activeExamAns[currentExamIdx] = optKey; renderExamQuestion(); }
        window.examNext = function() { if(currentExamIdx < activeExamQs.length - 1) { currentExamIdx++; renderExamQuestion(); } }
        window.examPrev = function() { if(currentExamIdx > 0) { currentExamIdx--; renderExamQuestion(); } }

        window.filterReview = function(status, btn) {
            document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
            document.querySelectorAll('.review-card').forEach(card => {
                if(status === 'all' || card.getAttribute('data-status') === status) card.style.display = 'block';
                else card.style.display = 'none';
            });
        }

        window.submitExam = function() {
            if(timeRemainingSecs > 0 && !confirm("Submit exam early?")) return;
            clearInterval(timerInterval);
            
            let correct = 0, wrong = 0, skipped = 0;
            activeExamQs.forEach((q, i) => {
                let sel = activeExamAns[i];
                if(sel) {
                    if(sel === q.ans) correct++;
                    else wrong++;
                } else skipped++;
            });
            
            let maxScore = 0, score = 0;
            if(examStyle === 'NEET') {
                maxScore = activeExamQs.length * 4;
                score = (correct * 4) - (wrong * 1);
            } else {
                maxScore = activeExamQs.length * 1;
                score = correct * 1; 
            }
            
            currentContext = 'exam-review';
            document.getElementById('header-quiz-name').innerText = "Exam Results";
            document.getElementById('lesson-reset-btn').style.display = 'none';
            
            let bannerHtml = `
                <div style="font-size:0.9rem; font-weight:800; color:var(--text-muted); text-transform:uppercase; margin-bottom:10px;">Exam Score (${examStyle})</div>
                <div style="font-size:3rem; font-weight:800; color:var(--accent); margin-bottom:20px;">${score} <span style="font-size:1.2rem; color:var(--text-muted);">/ ${maxScore}</span></div>
                
                <div style="display:flex; justify-content:center; gap:10px; margin-bottom:25px; flex-wrap:wrap;">
                    <div style="background:var(--green-bg); color:var(--green); padding:8px 15px; border-radius:12px; font-weight:700;"><i class="fas fa-check"></i> Correct: ${correct}</div>
                    <div style="background:var(--red-bg); color:var(--red); padding:8px 15px; border-radius:12px; font-weight:700;"><i class="fas fa-times"></i> Incorrect: ${wrong}</div>
                    <div style="background:var(--bg); color:var(--text-muted); padding:8px 15px; border-radius:12px; font-weight:700; border:1px solid var(--border);"><i class="fas fa-minus"></i> Skipped: ${skipped}</div>
                </div>

                <div class="result-kind-nudge">
                    Knowledge becomes powerful when it makes you useful to someone else.
                </div>

                <div style="display:flex; gap:10px; overflow-x:auto; padding-bottom:5px; justify-content:center;">
                    <button class="filter-btn active" onclick="filterReview('all', this)">All</button>
                    <button class="filter-btn" onclick="filterReview('correct', this)">Correct</button>
                    <button class="filter-btn" onclick="filterReview('wrong', this)">Incorrect</button>
                    <button class="filter-btn" onclick="filterReview('skipped', this)">Skipped</button>
                </div>
            `;
            document.getElementById('exam-result-banner').innerHTML = bannerHtml;
            document.getElementById('exam-result-banner').style.display = 'block';
            
            let html = '';
            activeExamQs.forEach((q, idx) => {
                let sel = activeExamAns[idx];
                let isBm = bookmarks[q.id] ? 'active' : '';
                let imgHtml = q.img ? `<img src="${q.img}" class="q-img">` : '';
                
                let qStatus = 'skipped';
                if(sel) {
                    if(sel === q.ans) qStatus = 'correct';
                    else qStatus = 'wrong';
                }
                
                html += `
                <div class="q-card review-card" data-status="${qStatus}" id="card-${q.id}">
                    <button class="bookmark-btn ${isBm}" onclick="toggleBookmark('${q.id}', this)"><i class="fas fa-bookmark"></i></button>
                    <div class="q-text"><span style="color:var(--accent);">Q${idx+1}.</span> ${q.text} ${imgHtml}</div>
                    <div class="opts-container locked">`;
                
                for(let optKey in q.options) {
                    let cClass = '';
                    if (optKey === q.ans) cClass = 'correct';
                    else if (optKey === sel && sel !== q.ans) cClass = 'wrong';

                    html += `
                        <div class="option ${cClass}">
                            <div class="opt-lbl">${optKey}</div>
                            <div style="flex:1;">${q.options[optKey]}</div>
                        </div>`;
                }
                
                html += `
                    </div>
                    <div class="expl" style="display:block;">
                        <strong style="color:var(--green); font-size:1.05rem;"><i class="fas fa-check-circle"></i> Correct Answer: ${q.ans}</strong><br><br>
                        ${q.expl}
                    </div>
                </div>`;
                
                if(qStatus === 'wrong') mistakes[q.id] = true;
            });
            
            saveState();
            document.getElementById('quiz-container').innerHTML = html;
            showScreen('screen-quiz');
        }

        init();
    </script>
